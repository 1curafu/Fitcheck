+ *

+ */

+  generatedOn: string,
+  predicted: string,
+  chosen: string,
+): Promise<void> {
+  const supabase = await createClient();
+  await supabase
+    .from("occasion_overrides")
+    .insert({ user_id: userId, generated_on: generatedOn, predicted, chosen });
