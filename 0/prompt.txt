+import { buildCandidates } from "../candidates";
+
+/**
+ * Coverage measurement for candidate generation.
+ *
+ * Reported symptom: the generator repeats pieces across the day's looks and
+ * never uses most of the closet. These tests measure what the candidate set can
+ * actually reach, so a fix has a before/after number rather than a feeling.
+ */
+
+/** A realistic closet: n of each required category, all mutually compatible. */
+function closet(n: number) {
+  const mk = (cat: string, i: number) => ({
+    id: `${cat}-${i}`,
+    category: cat,
+    colors: ["navy"],
+    formality: 3,
+    seasons: [] as string[],
+    material: "cotton",
+  });
+  return [
+    ...Array.from({ length: n }, (_, i) => mk("Tops", i)),
+    ...Array.from({ length: n }, (_, i) => mk("Bottoms", i)),
+    ...Array.from({ length: n }, (_, i) => mk("Shoes", i)),
+    ...Array.from({ length: 3 }, (_, i) => mk("Outerwear", i)),
+    ...Array.from({ length: 3 }, (_, i) => mk("Accessories", i)),
+  ];
+}
+
+const args = {
+  band: [1, 5] as [number, number],
+  weather: { tempC: 18, rain: false },
+  season: "Summer",
+  excludeItemIds: [] as string[],
+  maxAccessories: 1,
+};
+
+function reach(combos: { category: string; id: string }[][], cat: string): number {
+  return new Set(combos.flat().filter((i) => i.category === cat).map((i) => i.id)).size;
+}
+
+test("MEASUREMENT: every eligible top is reachable", () => {
+  const combos = buildCandidates(closet(20), args);

+  expect(reach(combos, "Tops")).toBe(20);
